            <input type="password" id="apiKeyInput" placeholder="Enter MDBlist API Key">

            <button onclick="saveApiKey()">Save & Load Watchlist</button>

            <p>Connecting to your live watchlist...</p>

    <script>

        let cachedWatchlist = [];

        window.onload = function() {

            const savedKey = localStorage.getItem("mdblist_api_key");

            if (savedKey) {

                document.getElementById("apiKeyInput").value = savedKey;

                document.getElementById("settings-box").style.display = "none";

                fetchWatchlist(savedKey);

            }

        };

        function saveApiKey() {

            const key = document.getElementById("apiKeyInput").value.trim();

            if (key) {

                localStorage.setItem("mdblist_api_key", key);

                document.getElementById("settings-box").style.display = "none";

                fetchWatchlist(key);

            }

        }

        async function fetchWatchlist(apiKey) {

            const container = document.getElementById("movie-list");

            container.innerHTML = "<p>Connecting to MDBlist API...</p>";

            try {

                const response = await fetch(`https://api.mdblist.com/watchlist/items?apikey=${apiKey}`);

                if (!response.ok) throw new Error("Failed to fetch watchlist. Check your API key.");

               

                const data = await response.json();

               

                // Strictly filter for movies only

                cachedWatchlist = data.filter(item => item.mediatype === 'movie');

                if (cachedWatchlist.length === 0) {

                    container.innerHTML = "<p>No movies found in your watchlist.</p>";

                    return;

                }

                document.getElementById("pick-btn").style.display = "inline-block";

                pickRandomMovies();

            } catch (error) {

                container.innerHTML = `<p style="color: #ff6b6b;">Error: ${error.message}</p>`;

                document.getElementById("settings-box").style.display = "block";

            }

        }

        function pickRandomMovies() {

            if (cachedWatchlist.length === 0) return;

            let shuffled = [...cachedWatchlist].sort(() => 0.5 - Math.random());

            let selected = shuffled.slice(0, 3);

            let container = document.getElementById("movie-list");

            container.innerHTML = "";

            selected.forEach((movie, index) => {

                let releaseYear = movie.release_year || (movie.release_date ? movie.release_date.split("-")[0] : "N/A");

                let synopsis = movie.description || "A captivating feature film handpicked from your live watchlist.";

               

                container.innerHTML += `

                    <div class="panel">

                        <h3>${index + 1}. ${movie.title} (${releaseYear})</h3>

                        <p style="font-size: 0.85rem; line-height: 1.4;">${synopsis}</p>

                    </div>

                `;

            });

        }
